import assert from 'node:assert/strict';
import {readFileSync,writeFileSync,mkdirSync,copyFileSync,existsSync} from 'node:fs';
import {join,dirname} from 'node:path';
import {pathToFileURL} from 'node:url';
import {spawnSync} from 'node:child_process';
const w=process.argv[2], dirs=JSON.parse(readFileSync(join(w,'host66/installed-directories.private.json'),'utf8'));
const core=dirname(dirname(dirs['filmcraft-use']));
const {createReviewRequest,evaluateReview,reviewRequestHash}=await import(pathToFileURL(join(core,'src/quality/review_contract.ts')).href);
const {reviewerIdentity}=await import(pathToFileURL(join(core,'src/quality/review_service.ts')).href);
const {sha256,stableJson}=await import(pathToFileURL(join(core,'src/support/json.ts')).href);
const root=join(w,'fixed66-proxy-only');mkdirSync(root);
const source=join(w,'fixed66-chinese/v1');const saved=sha256(readFileSync(join(source,'project.fcproj')));
for(const name of ['project.fcproj','frame-0000.png','captions.srt'])copyFileSync(join(source,name),join(root,name));
assert.equal(existsSync(join(root,'film.mp4')),false);
const tools={ffmpeg:'/opt/homebrew/bin/ffmpeg',ffprobe:'/opt/homebrew/bin/ffprobe',native:join(w,'plugin65-sequence-audit/empty-public-runtime/filmcraft/0.2.0-craft.5/filmcraft-cli')};
const command={id:'filmcraft-independent-media',version:'1',executable:'/opt/anaconda3/bin/python3',script:join(core,'scripts/review_media.py'),tools};
const files=Object.fromEntries(['project.fcproj','frame-0000.png','captions.srt'].map(n=>{const b=readFileSync(join(root,n));return[n,{sha256:sha256(b),bytes:b.length}]}));
const criteria={goal:{text:'Native project and preview only, no formal export',constraints:['formal movie absent'],audioRequired:true,expectedDurationMs:3000},rubric:{id:'proxy-only',version:'1',required:['decode','videoTiming','audioClipping','subtitleTiming'],thresholds:{syncToleranceMs:50,clippingAmplitude:.999,clippingFraction:.001}},reviewer:reviewerIdentity(command)};
const request=createReviewRequest({taskId:'fixed66-proxy-only',attemptId:'one',epoch:1,bindingSha256:saved,candidateSha256:sha256(stableJson(files)),projectSha256:saved,inputHashes:{},executionSha256:saved,files},criteria);
writeFileSync(join(root,'request.json'),stableJson(request)+'\n');writeFileSync(join(root,'tools.json'),stableJson(tools));
const run=spawnSync(command.executable,['-I','-B',command.script,'--request',join(root,'request.json'),'--artifacts',root,'--tools',join(root,'tools.json')],{encoding:'utf8',timeout:60000});assert.equal(run.status,0,run.stderr);
const wire=JSON.parse(run.stdout);assert.equal(wire.requestSha256,reviewRequestHash(request));const result=evaluateReview(request,wire.assessment);
assert.equal(result.engineeringAcceptance,'PASS');assert.equal(result.technicalAcceptance,'NOT_RUN');assert.equal(result.decision,'manual_review');assert.equal(result.dimensions.decode.status,'NOT_RUN');assert.equal(result.userAcceptance,'NOT_RUN');assert.equal(sha256(readFileSync(join(source,'project.fcproj'))),saved);
writeFileSync(join(root,'report.private.json'),JSON.stringify({result:'PASS',fixedRef:'v0.1.0-dev.66',sourceRef:'v0.1.0-dev.49',installedCore:core,files,reviewer:criteria.reviewer,assessment:wire.assessment,reviewResult:result,sourcePreserved:true,formalMovieExists:false},null,2)+'\n');console.log('PASS: real native reopen plus preview retains formal technical NOT_RUN');
